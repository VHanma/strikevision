        clearInterval(testTimer);

        $("startTest").textContent="START TEST";

      }
    },1000);
  } else if (activeMode === "combo" || activeMode === "technique") {

  }
});

function drawSpark(samples) {
  const c=$("sparkline"),x=c.getContext("2d"),w=c.width,h=c.height;
  x.clearRect(0,0,w,h);x.fillStyle="#0d0f0f";x.fillRect(0,0,w,h);
  if(!samples?.length)return;
  const max=Math.max(...samples.map(s=>s.v),1);
  x.strokeStyle="#c7ff19";x.lineWidth=5;x.beginPath();
  samples.forEach((s,i)=>{
    const px=(i/(samples.length-1||1))*w;
    const py=h-14-(s.v/max)*(h-28);
    i?x.lineTo(px,py):x.moveTo(px,py);
  });x.stroke();
  x.strokeStyle="rgba(255,255,255,.15)";x.lineWidth=2;x.beginPath();x.moveTo(0,h-14);x.lineTo(w,h-14);x.stroke();
}

function showResult(record) {
  showScreen("result");
  const previous = Store.getRecords().filter(r=>r.id!==record.id && !r.demo);
  const oldBest = previous.length ? Math.max(...previous.map(r=>r.mph)) : 0;
  const isPR = !record.demo && record.mph > oldBest;
  $("prBadge").classList.toggle("hidden",!isPR);
  $("resultEyebrow").textContent = record.demo ? "DEMO STRIKE • NOT A LIVE MEASUREMENT" : "VALID STRIKE CAPTURED";
  $("resultMph").textContent=record.mph.toFixed(1);
  $("resultFps").textContent=`${record.ftS.toFixed(1)} ft/s`;
  $("resultType").textContent=record.type;
  $("resultSide").textContent=record.side.toUpperCase();
  $("resultForce").textContent=`${Math.round(record.forceLow)}–${Math.round(record.forceHigh)}`;
  $("resultEnergy").textContent=record.energyFtLb.toFixed(1);
  $("resultMomentum").textContent=record.momentumLbS.toFixed(2);
  $("resultQuality").textContent=record.quality;
  $("resultConfidence").textContent=`${Math.round(record.confidence*100)}% CONF`;
  $("reactionResult").classList.toggle("hidden",record.reactionMs==null);
  if(record.reactionMs!=null)$("reactionMs").textContent=record.reactionMs;
  $("whyBody").innerHTML=`
    Peak fist speed came from the highest smoothed wrist velocity during one complete punch event.
    The event traveled about <b>${record.displacementIn}"</b> and tracking confidence was <b>${Math.round(record.confidence*100)}%</b>.
    Force uses <b>${record.effectiveWeightLb} lb</b> effective striking weight (${settings.massPct}% of body weight) and an impact-time window of
    <b>${settings.impactLowS.toFixed(3)}–${settings.impactHighS.toFixed(3)} s</b>. It is an impulse-based estimate, not a sensor-measured impact force.

  drawSpark(record.samples);
  $("againBtn").onclick=()=>{activeMode=record.mode;openCameraMode()};
}

$("captureScale").addEventListener("click", async ()=>{
  saveAllSettings();
  if(!currentLandmarks){
    toast("Open a camera mode first, stand in frame, then return and capture.");
    return;
  }
  const l=currentLandmarks;
  let best=null;
  for(const side of ["left","right"]){
    const ids=HAND[side],a=mapPoint(l[ids.shoulder]),b=mapPoint(l[ids.wrist]);
    const px=dist(a,b), conf=Math.min(lmConf(l[ids.shoulder]),lmConf(l[ids.wrist]));
    if(conf>=settings.minConfidence && (!best||px>best.px)) best={side,px};
  }
  if(!best){toast("Could not see an extended arm clearly.");return}
  const reach=best.side==="left"?settings.leftReachIn:settings.rightReachIn;
  settings.inchesPerPixel=reach/best.px;
  Store.saveSettings(settings);
  $("scaleStatus").textContent=`${settings.inchesPerPixel.toFixed(4)} IN/PX`;
  toast(`${best.side.toUpperCase()} arm scale captured.`);
});
